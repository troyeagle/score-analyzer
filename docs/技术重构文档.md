## 一、重构目标

1. 正确解析 MusicXML 的多声部、多谱表结构
2. 支持钢琴等大谱表乐器的左右手显示

## 二、数据模型重构

### 2.1 新的数据类型定义

```typescript
// 核心解析结果
interface MusicXMLParseResult {
  metadata: ScoreMetadata
  parts: Part[]
}

// 声部
interface Part {
  id: string
  name: string
  abbreviation: string
  staves: number           // 谱表数量（钢琴=2，其他=1）
  instruments: Instrument[]
  measures: Measure[]      // 该声部的所有小节
}

// 乐器信息
interface Instrument {
  id: string
  name: string
  sound?: string
}

// 小节
interface Measure {
  id: string
  number: number
  width?: number
  attributes: MeasureAttributes
  voices: Map<number, Voice>  // 按 voice 编号组织
  directions: Direction[]
  barlines: Barline[]
}

// 声部（voice）
interface Voice {
  id: number               // voice 编号 (1, 2, 3, 4)
  staff: number            // 谱表编号 (1, 2)
  notes: Note[]
}

// 音符
interface Note {
  id: string
  pitch: string            // 如 "C#4"
  duration: number
  type: string             // whole, half, quarter, eighth, 16th
  stem: string             // up, down
  beam?: string            // begin, continue, end
  accidental?: string      // sharp, flat, natural, double-sharp
  dots: number
  voice: number            // 所属 voice
  staff: number            // 所属谱表
  isChord: boolean         // 是否为和弦（与前一个音符同时发声）
  isRest: boolean          // 是否为休止符
  tie?: 'start' | 'stop' | 'continue'
  lyrics: Lyric[]
  notations: Notation[]
}

// 歌词
interface Lyric {
  number: string           // 如 "part1verse1"
  syllabic: string         // single, begin, middle, end
  text: string
  extend?: boolean         // 延音线
}

// 记号
interface Notation {
  tied?: { type: string }
  slur?: { type: string, number: number }
  fermata?: string
  articulations?: string[]
}

// 小节属性
interface MeasureAttributes {
  key: string
  mode: string
  time: string
  divisions: number
  staves: number
  clefs: Clef[]           // 支持多个谱号
}

// 谱号
interface Clef {
  number: number           // 谱表编号
  sign: string             // G, F, C
  line: number
  clefOctaveChange?: number
}

// 方向标记
interface Direction {
  type: string             // metronome, dynamics, words
  content: any
  staff?: number
  voice?: number
}

// 小节线
interface Barline {
  location: string         // left, right
  barStyle: string         // regular, heavy, light-light, light-heavy
}
```

### 2.2 数据模型优势

| 特性 | 旧模型 | 新模型 |
|------|--------|--------|
| 小节归属 | 扁平数组，丢失声部信息 | 按声部层次组织 |
| 多谱表 | 不支持 | 通过 staff 字段区分 |
| 多声部 | 不支持 | 通过 voice 和 backup 处理 |
| 歌词 | 不支持 | 完整支持 |
| 扩展性 | 差 | 好，易于添加新功能 |

### 3.2 关键解析逻辑

#### 3.2.1 多谱表解析

```typescript
private extractMeasureAttributes(element: Element): MeasureAttributes {
  const attributesElement = element.querySelector('attributes')
  if (!attributesElement) return DEFAULT_ATTRIBUTES
  
  // 解析谱表数量
  const stavesElement = attributesElement.querySelector('staves')
  const staves = stavesElement ? parseInt(stavesElement.textContent || '1') : 1
  
  // 解析每个谱表的谱号
  const clefs: Clef[] = []
  const clefElements = attributesElement.querySelectorAll('clef')
  clefElements.forEach(clefElement => {
    const number = parseInt(clefElement.getAttribute('number') || '1')
    const sign = clefElement.querySelector('sign')?.textContent || 'G'
    const line = parseInt(clefElement.querySelector('line')?.textContent || '2')
    const octaveChange = parseInt(clefElement.querySelector('clef-octave-change')?.textContent || '0')
    
    clefs.push({ number, sign, line, clefOctaveChange: octaveChange })
  })
  
  return { staves, clefs, ... }
}
```

#### 3.2.2 多声部和 Backup 解析

```typescript
private extractVoices(measureElement: Element): Map<number, Voice> {
  const voices = new Map<number, Voice>()
  let currentStaff = 1
  
  const elements = measureElement.children
  let i = 0
  
  while (i < elements.length) {
    const element = elements[i]
    
    if (element.tagName === 'note') {
      const note = this.extractNoteData(element)
      const voiceId = note.voice
      
      if (!voices.has(voiceId)) {
        voices.set(voiceId, {
          id: voiceId,
          staff: note.staff,
          notes: []
        })
      }
      
      voices.get(voiceId)!.notes.push(note)
    }
    else if (element.tagName === 'backup') {
      // backup 不创建音符，只是标记位置回退
      // 实际处理在渲染时进行
    }
    
    i++
  }
  
  return voices
}
```

## 四、渲染引擎重构

### 4.1 多谱表布局

```typescript
class LayoutEngine {
  // 计算多谱表布局
  layout(score: MusicXMLParseResult): StaveLayout[] {
    const layouts: StaveLayout[] = []
    
    score.parts.forEach(part => {
      if (part.staves > 1) {
        // 大谱表（如钢琴）
        layouts.push(this.layoutGrandStaff(part))
      } else {
        // 单谱表
        layouts.push(this.layoutSingleStaff(part))
      }
    })
    
    return layouts
  }
  
  // 大谱表布局
  private layoutGrandStaff(part: Part): StaveLayout {
    return {
      type: 'grand',
      staves: [
        { type: 'treble', y: 0 },
        { type: 'bass', y: 80 }  // 间距可配置
      ],
      brace: true
    }
  }
}
```
